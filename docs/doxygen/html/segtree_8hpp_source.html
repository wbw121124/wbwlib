85  T resl = id_, resr = id_;
86  for (l += n_ - 1, r += n_ - 1; l <= r; l >>= 1, r >>= 1) {
87  if (l & 1) resl = op_(resl, tr_[l++]);
88  if (!(r & 1)) resr = op_(tr_[r--], resr);
89  }

91  }

112 template<class T = i64, bool UseMax = true>
113 class LazySeg {
114  struct Node {
115  T sum, mx;
116  T lz_add, lz_set;
117  bool has_set;
118  Node() : sum(T()), mx(T()), lz_add(T()), lz_set(T()), has_set(false) {}
119  };
120  int n_;
121  std::vector<Node> t_;
122 
124  void apply_add(int p, int len, const T& v) {
125  t_[p].sum += v * T(len);
126  if (UseMax) t_[p].mx += v;
127  t_[p].lz_add += v;
128  }
129 
131  void apply_set(int p, int len, const T& v) {
132  t_[p].sum = v * T(len);
133  if (UseMax) t_[p].mx = v;
134  t_[p].has_set = true;
135  t_[p].lz_set = v;
136  t_[p].lz_add = T(); // 赋值覆盖旧加法
137  }
138 
140  void push(int p, int len) {
141  if (len <= 1) return;
142  int lc = p << 1, rc = p << 1 | 1;
143  int hl = len >> 1, hr = len - hl;
144  if (t_[p].has_set) {
145  apply_set(lc, hl, t_[p].lz_set);
146  apply_set(rc, hr, t_[p].lz_set);
147  t_[p].has_set = false;
148  }
149  if (t_[p].lz_add != T()) {
150  apply_add(lc, hl, t_[p].lz_add);
151  apply_add(rc, hr, t_[p].lz_add);
152  t_[p].lz_add = T();
153  }
154  }
155 
157  void pull(int p) {
158  t_[p].sum = t_[p << 1].sum + t_[p << 1 | 1].sum;
159  if (UseMax) t_[p].mx = (std::max)(t_[p << 1].mx, t_[p << 1 | 1].mx);
160  }
161 
163  void build(int p, int l, int r, const std::vector<T>& a) {
164  t_[p] = Node();
165  if (l == r) { t_[p].sum = t_[p].mx = a[l]; return; }
166  int mid = (l + r) >> 1;
167  build(p << 1, l, mid, a);
168  build(p << 1 | 1, mid + 1, r, a);
169  pull(p);
170  }
171 
173  void update(int p, int l, int r, int ql, int qr, const T& v, bool is_assign) {
174  if (ql <= l && r <= qr) {
175  if (is_assign) apply_set(p, r - l + 1, v);
176  else apply_add(p, r - l + 1, v);
177  return;
178  }
179  push(p, r - l + 1);
180  int mid = (l + r) >> 1;
181  if (ql <= mid) update(p << 1, l, mid, ql, qr, v, is_assign);
182  if (qr > mid) update(p << 1 | 1, mid + 1, r, ql, qr, v, is_assign);
183  pull(p);
184  }
185 
187  T qsum(int p, int l, int r, int ql, int qr) {
188  if (ql <= l && r <= qr) return t_[p].sum;
189  push(p, r - l + 1);
190  int mid = (l + r) >> 1;
191  T res = T();
192  if (ql <= mid) res += qsum(p << 1, l, mid, ql, qr);
193  if (qr > mid) res += qsum(p << 1 | 1, mid + 1, r, ql, qr);
194  return res;
195  }
196 
198  T qmax(int p, int l, int r, int ql, int qr) {
199  if (ql <= l && r <= qr) return t_[p].mx;
200  push(p, r - l + 1);
201  int mid = (l + r) >> 1;
202  T res = T(-std::numeric_limits<i64>::max());
203  if (ql <= mid) res = (std::max)(res, qmax(p << 1, l, mid, ql, qr));
204  if (qr > mid) res = (std::max)(res, qmax(p << 1 | 1, mid + 1, r, ql, qr));
205  return res;
206  }

213  explicit LazySeg(int n) : n_(n), t_(4 * n + 4) {}
214 
219  void build(const std::vector<T>& a) { build(1, 1, n_, a); }
220 
227  void add(int l, int r, const T& v) { if (l <= r) update(1, 1, n_, l, r, v, false); }
228 
235  void assign(int l, int r, const T& v) { if (l <= r) update(1, 1, n_, l, r, v, true); }
236 
243  T sum(int l, int r) {
244  if (l > r) return T();
245  return qsum(1, 1, n_, l, r);
246  }

254  T max(int l, int r) {
255  static_assert(UseMax, "LazySeg<UseMax=false> 不支持 max 查询");

257  return qmax(1, 1, n_, l, r);
