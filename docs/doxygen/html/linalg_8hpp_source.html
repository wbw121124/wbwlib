32 template<class T>
33 inline enable_if_t<std::is_arithmetic<T>::value, int>
34 gauss_impl(std::vector<std::vector<T>>& a, double fp_eps) {
35  int n = (int)a.size(), m = (int)a[0].size();
36  int rank = 0;
37  for (int col = 0; col < m - 1 && rank < n; ++col) {
38  int piv = rank;
39  double best = (fp_eps > 0) ? fp_eps : 1e-9;
40  for (int i = rank; i < n; ++i) {
41  double cur = std::fabs(static_cast<double>(a[i][col]));
42  if (cur > best) { best = cur; piv = i; }
43  }
44  if (std::fabs(static_cast<double>(a[piv][col])) <= best) continue; // 整列为 0 → 自由列
45  std::swap(a[piv], a[rank]);
46  T div = a[rank][col];
47  for (int j = col; j < m; ++j) a[rank][j] = a[rank][j] / div;
48  for (int i = 0; i < n; ++i) {
49  if (i == rank) continue;
50  T coef = a[i][col];
51  if (coef == T()) continue;
52  for (int j = col; j < m; ++j) a[i][j] = a[i][j] - coef * a[rank][j];
53  }
54  ++rank;
55  }
56  return rank;
57 }

112 template<class T>
113 int solve_linear(std::vector<std::vector<T>> A, std::vector<T> b, std::vector<T>& x) {
114  int n = (int)A.size();
115  WBWLIB_ASSERT(n > 0);
116  for (int i = 0; i < n; ++i) A[i].push_back(b[i]); // 增广
117  int rank = gauss_elim(A, 0.0); // 精确消元（模意义下）
118  // 检查无解：存在全零行但常数列非零
119  for (int i = 0; i < n; ++i) {
120  bool allz = true;
121  for (int j = 0; j < n; ++j) if (A[i][j] != T()) { allz = false; break; }
122  if (allz && A[i][n] != T()) return -1;
123  }
124  if (rank < n) return 1; // 无穷解
125  x.assign(n, T());
126  for (int i = 0; i < n; ++i) x[i] = A[i][n];
127  return 0;
128 }

138 template<class T>
139 T det(std::vector<std::vector<T>> a) {
140  int n = (int)a.size();
141  T d = T(1);
142  for (int col = 0; col < n; ++col) {
143  int piv = col;
144  while (piv < n && a[piv][col] == T()) ++piv;
145  if (piv == n) return T(0);
146  if (piv != col) { std::swap(a[piv], a[col]); d = -d; }
147  T div = a[col][col];
148  d = d * div;
149  for (int i = col + 1; i < n; ++i) {
150  T coef = a[i][col];
151  if (coef == T()) continue;
152  for (int j = col; j < n; ++j)
153  a[i][j] = a[i][j] - coef / div * a[col][j];
154  }
155  }
156  return d;
157 }
