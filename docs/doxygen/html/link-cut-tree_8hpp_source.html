43  bool is_root(int x) const {
44  return ch0_[fa_[x]] != x && ch1_[fa_[x]] != x;
45  }
47  void push_up(int x) {
48  siz_[x] = siz_[ch0_[x]] + siz_[ch1_[x]] + 1;
49  sum_[x] = sum_[ch0_[x]] + sum_[ch1_[x]] + val_[x];
50  }
52  void apply_add(int x, T v) {
53  if (!x) return;
54  val_[x] += v;
55  sum_[x] += v * siz_[x];
56  add_[x] += v;
57  }
59  void apply_rev(int x) {
60  if (!x) return;
61  std::swap(ch0_[x], ch1_[x]);
62  rev_[x] ^= 1;
63  }
65  void push_down(int x) {
66  if (rev_[x]) {
67  apply_rev(ch0_[x]);
68  apply_rev(ch1_[x]);
69  rev_[x] = 0;
70  }
71  if (add_[x] != T()) {
72  apply_add(ch0_[x], add_[x]);
73  apply_add(ch1_[x], add_[x]);
74  add_[x] = T();
75  }
76  }
78  void rotate(int x) {
79  int y = fa_[x], z = fa_[y];
80  int k = (ch1_[y] == x); // 1 表示 x 为 y 的右孩子
81  if (!is_root(y)) {
82  if (ch0_[z] == y) ch0_[z] = x;
83  else ch1_[z] = x;
84  }
85  if (k) {
86  ch1_[y] = ch0_[x];
87  if (ch0_[x]) fa_[ch0_[x]] = y;
88  ch0_[x] = y;
89  } else {
90  ch0_[y] = ch1_[x];
91  if (ch1_[x]) fa_[ch1_[x]] = y;
92  ch1_[x] = y;
93  }
94  fa_[y] = x;
95  fa_[x] = z;
96  push_up(y);
97  push_up(x);
98  }

115  void splay(int x) {
116  std::vector<int> st;
117  st.push_back(x);
118  for (int y = x; !is_root(y); y = fa_[y]) st.push_back(fa_[y]);
119  for (int i = (int)st.size() - 1; i >= 0; --i) push_down(st[i]);
120  while (!is_root(x)) {
121  int y = fa_[x], z = fa_[y];
122  if (!is_root(y)) {
123  bool same = ((ch1_[y] == x) == (ch1_[z] == y));
124  if (same) rotate(y);
125  else rotate(x);
126  }
127  rotate(x);
128  }
129  }

136  int last = 0;
137  for (int y = x; y; y = fa_[y]) {
138  splay(y);
139  ch1_[y] = last;

142  }

144  }
