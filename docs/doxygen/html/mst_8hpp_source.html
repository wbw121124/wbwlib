63 inline MstResult<i64> kruskal(int n,
64  const std::vector<std::array<int, 3>>& edges) {
65  MstResult<i64> res;
66  std::vector<int> fa(n + 1), rnk(n + 1, 0);
67  for (int i = 1; i <= n; ++i) fa[i] = i;
68  std::function<int(int)> find = [&](int x) {
69  while (fa[x] != x) x = fa[x] = fa[fa[x]];
70  return x;
71  };
72  auto es = edges;
73  std::sort(es.begin(), es.end(),
74  [](const std::array<int, 3>& a, const std::array<int, 3>& b) {
75  return a[2] < b[2];
76  });
77  int cnt = 0;
78  for (auto& e : es) {
79  int u = find(e[0]), v = find(e[1]);
80  if (u == v) continue;
81  res.edges.push_back({e[0], e[1]});
82  res.total += e[2];
83  if (rnk[u] < rnk[v]) std::swap(u, v);
84  fa[v] = u;
85  if (rnk[u] == rnk[v]) ++rnk[u];
86  if (++cnt == n - 1) break;
87  }
88  res.ok = (cnt == n - 1);
89  return res;
90 }

108 template<class W>
109 i64 prim(int n, const WAdj<W>& g) {
110  const i64 INF = mst_INF();
111  std::vector<i64> key(n + 1, INF);
112  std::vector<char> used(n + 1, 0);
113  typedef std::pair<i64, int> P;
114  std::priority_queue<P, std::vector<P>, std::greater<P>> pq;
115  key[1] = 0;
116  pq.push({0, 1});
117  i64 total = 0;
118  int cnt = 0;
119  while (!pq.empty()) {
120  int u = pq.top().second;
121  i64 d = pq.top().first;
122  pq.pop();
123  if (used[u]) continue;
124  used[u] = 1;
125  total += d;
126  ++cnt;
127  for (auto& e : g.g[u])
128  if (!used[e.first] && e.second < key[e.first]) {
129  key[e.first] = e.second;
130  pq.push({key[e.first], e.first});
131  }
132  }
133  return cnt == n ? total : -1;
134 }
