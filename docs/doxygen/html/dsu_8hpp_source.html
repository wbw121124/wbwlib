153 class WeightedDSU {
154  std::vector<int> fa_;
155  std::vector<i64> wt_;
156  bool mod_mode_;
157  i64 mod_;
158 
159  public:
166  explicit WeightedDSU(int n, bool mod = false, i64 modulus = 0)
167  : mod_mode_(mod), mod_(modulus) {
168  fa_.resize(n + 1);
169  wt_.assign(n + 1, 0);
170  for (int i = 1; i <= n; ++i) fa_[i] = i;
171  }
172 
178  int find(int x) {
179  if (fa_[x] == x) return x;
180  int r = find(fa_[x]);
181  wt_[x] += wt_[fa_[x]]; // 路径压缩累加权值
182  fa_[x] = r;
183  return r;
184  }
185 
191  i64 to_root(int x) { find(x); return wt_[x]; }
192 
202  bool unite(int a, int b, i64 diff) {
203  int ra = find(a), rb = find(b);
204  if (ra == rb) {
205  i64 cur = to_root(b) - to_root(a);
206  if (mod_mode_) cur = norm(cur);
207  return cur == (mod_mode_ ? norm(diff) : diff);
208  }
209  // diff = value(b) - value(a) = (wt_b + value(rb)) - (wt_a + value(ra))
210  // 令 value(ra) = value(rb) + (wt_b + value(rb)... ) 化解：
211  i64 t = to_root(b) - to_root(a) - diff;
212  if (mod_mode_) t = norm(t);
213  fa_[ra] = rb;
214  wt_[ra] = t; // value(ra) = value(rb) + t
215  return true;
216  }
217 
224  bool same(int a, int b) { return find(a) == find(b); }
225 
232  i64 diff(int a, int b) { // value(b) - value(a)
233  find(a); find(b);
234  i64 d = to_root(b) - to_root(a);
235  return mod_mode_ ? norm(d) : d;
236  }
237 
238  private:
240  i64 norm(i64 x) const {
241  x %= mod_;
242  return x < 0 ? x + mod_ : x;
243  }
244 };

280  if (ra == rb) return false;
281  if (sz_[ra] < sz_[rb]) std::swap(ra, rb);

304  }
