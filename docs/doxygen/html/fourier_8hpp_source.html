43 inline void fft(std::vector<std::complex<double>>& a, bool invert) {
44  int n = (int)a.size();
45  // 位逆序置换
46  for (int i = 1, j = 0; i < n; ++i) {
47  int bit = n >> 1;
48  for (; j & bit; bit >>= 1) j ^= bit;
49  j ^= bit;
50  if (i < j) std::swap(a[i], a[j]);
51  }
52  for (int len = 2; len <= n; len <<= 1) {
53  double ang = 2.0 * 3.14159265358979323846 / len * (invert ? -1 : 1);
54  std::complex<double> wlen(std::cos(ang), std::sin(ang));
55  for (int i = 0; i < n; i += len) {
56  std::complex<double> w(1.0, 0.0);
57  for (int j = 0; j < len / 2; ++j) {
58  std::complex<double> u = a[i + j], v = a[i + j + len / 2] * w;
59  a[i + j] = u + v;
60  a[i + j + len / 2] = u - v;
61  w *= wlen;
62  }
63  }
64  }
65  if (invert) for (int i = 0; i < n; ++i) a[i] /= n;
66 }

94 template<class M>
95 inline void ntt_core(std::vector<M>& a, bool invert, M root) {
96  int n = (int)a.size();
97  const i64 mod = (i64)M::mod_value();
98  for (int i = 1, j = 0; i < n; ++i) {
99  int bit = n >> 1;
100  for (; j & bit; bit >>= 1) j ^= bit;
101  j ^= bit;
102  if (i < j) std::swap(a[i], a[j]);
103  }
104  for (int len = 2; len <= n; len <<= 1) {
105  M wlen = root.pow((mod - 1) / len); // g^((mod-1)/len)
106  if (invert) wlen = wlen.inv();
107  for (int i = 0; i < n; i += len) {
108  M w = M(1);
109  for (int j = 0; j < len / 2; ++j) {
110  M u = a[i + j], v = a[i + j + len / 2] * w;
111  a[i + j] = u + v;
112  a[i + j + len / 2] = u - v;
113  w *= wlen;
114  }
115  }
116  }
117  if (invert) {
118  M inv_n = M(n).inv();
119  for (int i = 0; i < n; ++i) a[i] *= inv_n;
120  }
121 }
