68  if (pool_.empty()) pool_.push_back({-1, -1, T()}); // 0 号空叶子

70  return (int)pool_.size() - 1;

74  int upd(int p, int l, int r, int pos, const T& delta) {
75  int np = newnode();
76  if (p >= 0) pool_[np] = pool_[p]; // p 为 -1 表示空节点
77  if (l == r) {
78  pool_[np].sum += delta;
79  return np;
80  }
81  int mid = (l + r) >> 1;
82  if (pos <= mid)
83  pool_[np].l = upd(p >= 0 ? pool_[p].l : -1, l, mid, pos, delta);
84  else
85  pool_[np].r = upd(p >= 0 ? pool_[p].r : -1, mid + 1, r, pos, delta);
86  pool_[np].sum = (pool_[np].l >= 0 ? pool_[pool_[np].l].sum : T()) +
87  (pool_[np].r >= 0 ? pool_[pool_[np].r].sum : T());
88  return np;
89  }

92  int kth(int x, int y, int l, int r, int k) const { // x,y 两种版本同结构
93  if (l == r) return l;
94  int ll = (x >= 0 ? pool_[x].l : -1), lr = (y >= 0 ? pool_[y].l : -1);
95  int left = (lr >= 0 ? pool_[lr].sum : T()) -
96  (ll >= 0 ? pool_[ll].sum : T());
97  T lc = left; // 差集左子树 count
98  int mid = (l + r) >> 1;
99  if (lc >= k) return kth(ll, lr, l, mid, k);
100  return kth(x >= 0 ? pool_[x].r : -1, y >= 0 ? pool_[y].r : -1,
101  mid + 1, r, k - lc);
102  }
103 
105  T qcnt(int x, int y, int l, int r, int ql, int qr) const {
106  if (ql <= l && r <= qr)
107  return T((y >= 0 ? pool_[y].sum : T()) - (x >= 0 ? pool_[x].sum : T()));
108  int mid = (l + r) >> 1;
109  T res = T();
110  if (ql <= mid)
111  res += qcnt(x >= 0 ? pool_[x].l : -1, y >= 0 ? pool_[y].l : -1,
112  l, mid, ql, qr);
113  if (qr > mid)
114  res += qcnt(x >= 0 ? pool_[x].r : -1, y >= 0 ? pool_[y].r : -1,
115  mid + 1, r, ql, qr);
116  return res;
117  }
