107  int extend(int c) {
108  s_.push_back(c);
109  ++n_;
110  int cur = get_fail(last_);
111  int now;
112  if (node_[cur].next[c] == 0) {
113  now = sz_++;
114  node_.push_back({});
115  node_[now].len = node_[cur].len + 2;
116  if (node_[now].len == 1) {
117  node_[now].fail = 1;
118  } else {
119  int f = get_fail(node_[cur].fail);
120  node_[now].fail = node_[f].next[c];
121  }
122  node_[cur].next[c] = now;
123  } else {
124  now = node_[cur].next[c];
125  }
126  last_ = now;
127  return now;
128  }

134  void build(const std::basic_string<Char>& s) {
135  for (Char ch : s) {
136  int nd = extend(CharMap::id(ch));
137  ++node_[nd].cnt; // 该最长回文计数 +1
138  }
139  // 拓扑累加：按 len 降序，把 cnt 累加到 fail 上
140  std::vector<int> order(sz_);
141  for (int i = 0; i < sz_; ++i) order[i] = i;
142  std::sort(order.begin(), order.end(),
143  [&](int a, int b) { return node_[a].len > node_[b].len; });
144  for (int v : order)
145  if (v >= 2) node_[node_[v].fail].cnt += node_[v].cnt;
146  }
