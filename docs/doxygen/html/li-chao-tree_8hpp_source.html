114  T query(i64 x, i64 L, i64 R, int id) const {
115  T res = 0;
116  bool first = true;
117  while (true) {
118  if (seg_[id].has) {
119  T v = seg_[id].k * x + seg_[id].b;
120  if (first || good(res, v)) { res = v; first = false; }
121  }
122  if (L == R) break;
123  i64 mid = (L + R) >> 1;
124  if (x <= mid) { id = id << 1; R = mid; }
125  else { id = id << 1 | 1; L = mid + 1; }
126  }
