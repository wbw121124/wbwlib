116  void build_cnt() {
117  int sz = (int)st_.size();
118  std::vector<int> order(sz);
119  int mx = 0;
120  for (int i = 0; i < sz; ++i) mx = std::max(mx, st_[i].len);
121  std::vector<int> bucket(mx + 1, 0);
122  for (int i = 0; i < sz; ++i) ++bucket[st_[i].len];
123  for (int i = 1; i <= mx; ++i) bucket[i] += bucket[i - 1];
124  for (int i = sz - 1; i >= 0; --i) order[--bucket[st_[i].len]] = i;
125  // 从长到短累加 endpos
126  for (int i = sz - 1; i > 0; --i) {
