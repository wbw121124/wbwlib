61  void build(const std::string& s) {
62  int n = (int)s.size();
63  n_ = n;
64  int m = n + 1; // 含哨兵，共 m 个后缀对象
65  std::vector<int> a(m);
66  for (int i = 0; i < n; ++i) a[i] = (int)(unsigned char)s[i] + 1;
67  a[n] = 0; // 哨兵最小
68 
69  std::vector<int> rk(m), init_ord(m);
70  // ---- 初始按第一字符计数排序 ----
71  int K = 256;
72  for (int i = 0; i < m; ++i) init_ord[i] = i;
73  cnt_.assign(K + 1, 0);
74  for (int x : a) ++cnt_[x];
75  for (int i = 1; i <= K; ++i) cnt_[i] += cnt_[i - 1];
76  for (int i = m - 1; i >= 0; --i) init_ord[--cnt_[a[i]]] = i;
77  // 初始排名（字符类）
78  int cls = 0;
79  rk[init_ord[0]] = 0;
80  for (int i = 1; i < m; ++i) {
81  if (a[init_ord[i]] == a[init_ord[i - 1]]) rk[init_ord[i]] = cls;
82  else rk[init_ord[i]] = ++cls;
83  }
84  // ---- 倍增 ----
85  for (int k = 1; k < m && cls < m - 1; k <<= 1) {
86  const int K2 = cls;
87  std::vector<int> ord(m);
88  for (int i = 0; i < m; ++i) ord[i] = i;
89  // 先按第二关键字（i+k 处排名，越界视为 -1 排最前）稳定排序
90  std::vector<int> key2(m);
91  for (int i = 0; i < m; ++i) key2[i] = (i + k < m) ? rk[i + k] + 1 : 0;
92  cnt_.assign(K2 + 2, 0);
93  for (int x : key2) ++cnt_[x];
94  for (int i = 1; i <= K2 + 1; ++i) cnt_[i] += cnt_[i - 1];
95  for (int i = m - 1; i >= 0; --i) ord[--cnt_[key2[i]]] = i;
96  // 再按第一关键字 rk[i] 稳定排序
97  cnt_.assign(cls + 2, 0);
98  for (int i = 0; i < m; ++i) ++cnt_[rk[i]];
99  for (int i = 1; i <= cls + 1; ++i) cnt_[i] += cnt_[i - 1];
100  std::vector<int> tmp(m);
101  for (int i = m - 1; i >= 0; --i) tmp[--cnt_[rk[ord[i]]]] = ord[i];
102  ord.swap(tmp);
103  // 重新分配类
104  std::vector<int> nrk(m);
105  nrk[ord[0]] = 0;
106  int cnew = 0;
107  for (int i = 1; i < m; ++i) {
108  int x = ord[i - 1], y = ord[i];
109  if (rk[x] == rk[y] &&
110  ((x + k < m ? rk[x + k] : -1) == (y + k < m ? rk[y + k] : -1)))
111  nrk[y] = cnew;
112  else nrk[y] = ++cnew;
113  }
114  cls = cnew;
115  rk.swap(nrk);
116  }
117  // init_ord 已无意义，最终顺序存于 rk 对应的排序后数组；
118  // 重新生成最终顺序 ord（含哨兵）：
119  std::vector<int> ord2(m);
120  for (int i = 0; i < m; ++i) ord2[i] = i;
121  {
122  // 按 rk 排序即最终后缀顺序（哨兵排最前）
123  cnt_.assign(cls + 2, 0);
124  for (int i = 0; i < m; ++i) ++cnt_[rk[i]];
125  for (int i = 1; i <= cls + 1; ++i) cnt_[i] += cnt_[i - 1];
126  std::vector<int> tmp(m);
127  for (int i = m - 1; i >= 0; --i) tmp[--cnt_[rk[ord2[i]]]] = ord2[i];
128  ord2.swap(tmp);
129  }
130  // ord2[0] 为哨兵后缀
131  sa.assign(n + 1, 0);
132  rank.assign(n + 1, 0);
133  for (int p = 1; p <= n; ++p) {
134  int pos = ord2[p];
135  sa[p] = pos;
136  rank[pos] = p;
137  }
138  // ---- Kasai 求 height ----
139  height.assign(n + 1, 0);
140  int h = 0;
141  for (int i = 0; i < n; ++i) {
142  if (rank[i] == 1) continue;
143  int j = sa[rank[i] - 1];
144  while (i + h < n && j + h < n && s[i + h] == s[j + h]) ++h;
145  height[rank[i]] = h;
146  if (h) --h;
147  }
148  }

159  int lcp(int i, int j) const {
160  if (i == j) return n_ - i;
161  int ri = rank[i], rj = rank[j];
162  if (ri > rj) std::swap(ri, rj);
163  // 区间 height 最小值：[ri+1, rj]（朴素 O(n)；需要 O(1) 请配 ST 表）
164  int res = 0x7fffffff;
165  for (int p = ri + 1; p <= rj; ++p) res = std::min(res, height[p]);
166  return res;
167  }
