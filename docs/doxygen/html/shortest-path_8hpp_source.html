42 std::vector<i64> dijkstra(int n, int s, const WAdj<W>& g) {
43  std::vector<i64> dist(n + 1, INF_LL);
44  typedef std::pair<i64, int> P;
45  std::priority_queue<P, std::vector<P>, std::greater<P>> pq;
46  dist[s] = 0;
47  pq.push({0, s});
48  while (!pq.empty()) {
49  int u = pq.top().second;
50  i64 d = pq.top().first;
51  pq.pop();
52  if (d != dist[u]) continue;
53  for (auto& e : g.g[u]) {
54  int v = e.first;
55  if (dist[u] + e.second < dist[v]) {
56  dist[v] = dist[u] + e.second;
57  pq.push({dist[v], v});
58  }
59  }
60  }
61  return dist;
62 }

105 inline void floyd(std::vector<std::vector<i64>>& d) {
106  int n = (int)d.size() - 1;
107  for (int k = 1; k <= n; ++k)
108  for (int i = 1; i <= n; ++i) {
109  if (d[i][k] >= INF_LL) continue;

111  if (d[k][j] < INF_LL && d[i][k] + d[k][j] < d[i][j])
112  d[i][j] = d[i][k] + d[k][j];
113  }

126  std::deque<int> dq;
127  std::vector<i64> dist(n + 1, INF_LL);
128  dist[s] = 0;
129  dq.push_front(s);
130  while (!dq.empty()) {
131  int u = dq.front();
132  dq.pop_front();
133  for (auto& e : g.g[u]) {
134  i64 nd = dist[u] + e.second;
135  if (nd < dist[e.first]) {
136  dist[e.first] = nd;
137  if (e.second == 0) dq.push_front(e.first);
138  else dq.push_back(e.first);
139  }
140  }
141  }
142  return dist;
143 }

154 std::vector<i64> dag_shortest(int n, const WAdj<W>& g,
155  std::vector<int>& order) {
156  std::vector<int> indeg(n + 1, 0);
157  for (int u = 1; u <= n; ++u)
158  for (auto& e : g.g[u]) ++indeg[e.first];
159  std::queue<int> q;
160  for (int u = 1; u <= n; ++u)
161  if (indeg[u] == 0) q.push(u);
162  order.clear();
163  while (!q.empty()) {
164  int u = q.front();
165  q.pop();
166  order.push_back(u);
167  for (auto& e : g.g[u])
168  if (--indeg[e.first] == 0) q.push(e.first);
169  }
170  // 不支持负权处理；DP 求从 1 出发
171  std::vector<i64> dist(n + 1, INF_LL);
172  dist[1] = 0;
173  for (int u : order)
174  if (dist[u] < INF_LL)
175  for (auto& e : g.g[u])
176  if (dist[u] + e.second < dist[e.first])
177  dist[e.first] = dist[u] + e.second;
178  return dist;
179 }
