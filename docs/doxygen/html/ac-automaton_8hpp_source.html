83  void build() {
84  fail.assign(tr_.size(), 0);
85  std::queue<int> q;
86  for (int c = 0; c < ALPHA; ++c) {
87  if (tr_[0].nxt[c]) {
88  tr_[tr_[0].nxt[c]].fail = 0;
89  q.push(tr_[0].nxt[c]);
90  }
91  }
92  while (!q.empty()) {
93  int u = q.front(); q.pop();
94  fail[u] = tr_[u].fail;
95  tr_[u].e2 = tr_[u].cnt + tr_[tr_[u].fail].e2;
96  for (int c = 0; c < ALPHA; ++c) {
97  int v = tr_[u].nxt[c];
98  if (v) {
99  tr_[v].fail = tr_[tr_[u].fail].nxt[c];
100  q.push(v);
101  } else {
102  tr_[u].nxt[c] = tr_[tr_[u].fail].nxt[c];
103  }
104  }
