55  int q = (int)qs.size();
56  int B = (int)std::max(1, (int)std::sqrt((double)n));
57  std::vector<int> ord(q);
58  for (int i = 0; i < q; ++i) ord[i] = i;
59  std::sort(ord.begin(), ord.end(), [&](int x, int y) {
60  int bx = qs[x].first / B, by = qs[y].first / B;
61  if (bx != by) return bx < by;
62  // 奇偶优化
63  return (bx & 1) ? qs[x].second > qs[y].second
64  : qs[x].second < qs[y].second;
65  });
66  std::vector<Ans> res(q);
67  int L = 1, R = 0;
68  for (int id : ord) {
69  int ql = qs[id].first, qr = qs[id].second;
70  while (L > ql) add(--L);
71  while (R < qr) add(++R);
72  while (L < ql) del(L++);
73  while (R > qr) del(R--);
74  res[id] = get();
75  }
