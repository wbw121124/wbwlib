34 template<class M, bool inverse>
35 inline void fwt_core(std::vector<M>& a, int kind) {
36  int n = (int)a.size();
37  for (int len = 1; len < n; len <<= 1) {
38  for (int i = 0; i < n; i += len << 1) {
39  for (int j = 0; j < len; ++j) {
40  M &x = a[i + j], &y = a[i + j + len];
41  if (kind == 0) { // OR
42  if (!inverse) y += x; else y -= x;
43  } else if (kind == 1) { // AND
44  if (!inverse) x += y; else x -= y;
45  } else { // XOR
46  M u = x + y, v = x - y;
47  if (inverse) u /= M(2), v /= M(2);
48  x = u; y = v;
49  }
50  }
51  }
52  }
53 }
