82  void build(const std::string& s, u64 base = 131ULL) {
83  n_ = s.size(); base_ = base % MOD1;
84  h1_.assign(n_ + 1, 0); h2_.assign(n_ + 1, 0);
85  p1_.assign(n_ + 1, 1); p2_.assign(n_ + 1, 1);
86  for (size_t i = 0; i < n_; ++i) {
87  h1_[i + 1] = (hashingdetail::mulmod(h1_[i], base_, MOD1) + (u64)s[i]) % MOD1;
88  h2_[i + 1] = (hashingdetail::mulmod(h2_[i], base_, MOD2) + (u64)s[i]) % MOD2;
89  p1_[i + 1] = hashingdetail::mulmod(p1_[i], base_, MOD1);
90  p2_[i + 1] = hashingdetail::mulmod(p2_[i], base_, MOD2);
91  }
