71  int n = (int)dist.size();
72  if (n == 0) return 0;
73  unsigned full = 1u << n;
74  const i64 INF = (i64)4e18;
75  std::vector<std::vector<i64>> f(full, std::vector<i64>(n, INF));
76  f[1][0] = 0;
77  for (unsigned mask = 1; mask < full; ++mask) {
78  for (int j = 0; j < n; ++j) {
79  if (!(mask & (1u << j))) continue;
80  i64 d = f[mask][j];
81  if (d >= INF) continue;
82  for (int k = 0; k < n; ++k) {
83  if (mask & (1u << k)) continue;
84  f[mask | (1u << k)][k] =
85  std::min(f[mask | (1u << k)][k], d + dist[j][k]);
86  }
87  }
