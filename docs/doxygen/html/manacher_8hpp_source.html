37 inline std::pair<std::vector<int>, std::vector<int>> manacher(const std::string& s) {
38  int n = (int)s.size();
39  std::vector<int> d1(n), d2(n);
40  for (int i = 0, l = 0, r = -1; i < n; ++i) {
41  int k = (i > r) ? 1 : std::min(d1[l + r - i], r - i + 1);
42  while (0 <= i - k && i + k < n && s[i - k] == s[i + k]) ++k;
43  d1[i] = k--;
44  if (i + k > r) { l = i - k; r = i + k; }
45  }
46  for (int i = 0, l = 0, r = -1; i < n; ++i) {
47  int k = (i > r) ? 0 : std::min(d2[l + r - i + 1], r - i + 1);
48  while (0 <= i - k - 1 && i + k < n && s[i - k - 1] == s[i + k]) ++k;
49  d2[i] = k--;
50  if (i + k > r) { l = i - k - 1; r = i + k; }
51  }
52  return {d1, d2};
53 }
