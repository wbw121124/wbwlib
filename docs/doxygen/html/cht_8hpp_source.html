113  if (x->k == y->k) x->p = x->m > y->m ? INF : -INF;
114  else x->p = div_f(y->m - x->m, x->k - y->k);
115  return x->p >= y->p;
116  }

131  while (isect(y, z)) z = st_.erase(z);
132  if (x != st_.begin() && isect(--x, y)) isect(x, y = st_.erase(y));
133  while ((y = x) != st_.begin() && (--x)->p >= y->p)
