39 inline i64 bsgs(i64 a, i64 b, i64 p) {
40  a %= p; b %= p;
41  if (b == 1) return 0; // a^0 ≡ 1
42  if (a == 0) return b == 0 ? 1 : -1; // a^x ≡ 0 (mod p) 仅当 b=0 且 x>=1
43  i64 m = (i64)std::ceil(std::sqrt((double)p));
44  // 小步：记录 a^j 的 j（b * a^j -> j）
45  std::unordered_map<i64, i64> mp;
46  mp.reserve(m + 5);
47  i64 base = 1; // a^0
48  for (i64 j = 0; j < m; ++j) {
49  i64 key = mul_mod(base, b, p);
50  if (mp.find(key) == mp.end()) mp[key] = j;
51  base = mul_mod(base, a, p);
52  }
53  // 大步：枚举 i，求 a^{i*m}
54  i64 am = mul_mod(base, i64(1), p); // 此时 base == a^m
55  i64 cur = am;
56  for (i64 i = 1; i <= m; ++i) {
57  auto it = mp.find(cur);
58  if (it != mp.end()) return i * m - it->second;
59  cur = mul_mod(cur, am, p);
60  }
61  return -1; // 无解
62 }
