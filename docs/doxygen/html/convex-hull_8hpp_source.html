34 std::vector<Point<T>> convex_hull(std::vector<Point<T>> ps, bool keep_collinear = false) {
35  if (ps.size() <= 1) return ps;
36  std::sort(ps.begin(), ps.end());
37  ps.erase(std::unique(ps.begin(), ps.end()), ps.end());
38  if (ps.size() == 1) return ps;
39  int n = (int)ps.size();
40  std::vector<Point<T>> hull;
41  hull.reserve(2 * n);
42  // 下凸壳
43  for (int i = 0; i < n; ++i) {
44  while ((int)hull.size() >= 2) {
45  T o = orient(hull[hull.size() - 2], hull.back(), ps[i]);
46  if (o > 0 || (keep_collinear && o == 0)) break; // 左转或保留共线
47  hull.pop_back();
48  }
49  hull.push_back(ps[i]);
50  }
51  // 上凸壳
52  int lower = (int)hull.size();
53  for (int i = n - 2; i >= 0; --i) {
54  while ((int)hull.size() > lower) {
55  T o = orient(hull[hull.size() - 2], hull.back(), ps[i]);
56  if (o > 0 || (keep_collinear && o == 0)) break;
57  hull.pop_back();
58  }
59  hull.push_back(ps[i]);
60  }
61  hull.pop_back(); // 移除起点重复（首尾相接）
62  // 若所有点共线，上面会退化为一条链；hull 仍有效（含两端点）。
63  return hull;
64 }

77 bool point_in_convex(const std::vector<Point<T>>& hull, const Point<T>& p) {
78  int n = (int)hull.size();
79  if (n == 0) return false;
80  if (n == 1) return p == hull[0];
81  if (n == 2) return collinear(hull[0], hull[1], p) &&
82  p.x >= std::min(hull[0].x, hull[1].x) &&
83  p.x <= std::max(hull[0].x, hull[1].x) &&
84  p.y >= std::min(hull[0].y, hull[1].y) &&
85  p.y <= std::max(hull[0].y, hull[1].y);
86  if (orient(hull[0], hull[1], p) < 0) return false;
87  if (orient(hull[0], hull[n - 1], p) > 0) return false;
88  // 二分找到最后一个 orient(hull[0], hull[mid], p) >= 0
89  int l = 1, r = n - 1;
90  while (r - l > 1) {
91  int mid = (l + r) >> 1;
92  if (orient(hull[0], hull[mid], p) >= 0) l = mid;
93  else r = mid;
94  }
95  return orient(hull[l], hull[r], p) >= 0;
96 }
