48 inline CartesianTree cartesian_tree(const std::vector<T>& a, bool is_max = false) {
49  int n = (int)a.size() - 1;
50  CartesianTree ct;
51  ct.l.assign(n + 1, 0);
52  ct.r.assign(n + 1, 0);
53  ct.fa.assign(n + 1, 0);
54  std::vector<int> st;
55  for (int i = 1; i <= n; ++i) {
56  int last = 0;
57  Cmp cmp; // Cmp：st.back() 比 i "更靠顶" 时弹出
58  while (!st.empty()) {
59  bool popit;
60  if (is_max) popit = cmp(a[st.back()], a[i]); // 大根：栈顶值 < i 的值
61  else popit = a[st.back()] > a[i]; // 小根：栈顶值 > i 的值
62  if (!popit) break;
63  last = st.back();
64  st.pop_back();
65  }
66  if (!st.empty()) {
67  ct.fa[i] = st.back();
68  ct.r[st.back()] = i;
69  }
70  ct.l[i] = last;
71  if (last) ct.fa[last] = i;
72  st.push_back(i);
73  }
74  ct.root = st[0];
75  for (int x : st) {
76  if (x != ct.root && ct.fa[x] == 0) ct.fa[x] = st[0];
77  }
78  return ct;
79 }
