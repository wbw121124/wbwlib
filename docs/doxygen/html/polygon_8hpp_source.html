72  int n = (int)p.size();
73  bool inside = false;
74  for (int i = 0, j = n - 1; i < n; j = i++) {
75  // 边界测试
76  if (orient(p[j], p[i], q) == 0 &&
77  q.x >= std::min(p[i].x, p[j].x) && q.x <= std::max(p[i].x, p[j].x) &&
78  q.y >= std::min(p[i].y, p[j].y) && q.y <= std::max(p[i].y, p[j].y))
79  return 2;
80  bool a = p[i].y > q.y, b = p[j].y > q.y;
81  if (a != b) {
82  T t = (q.y - p[j].y) * (p[i].x - p[j].x) - (q.x - p[j].x) * (p[i].y - p[j].y);
83  if ((a && t < 0) || (!a && t > 0)) inside = !inside;
84  }
85  }
86  return inside ? 1 : 0;
87 }
