54  void pull(int p) {
55  Node &u = t_[p], &a = t_[p << 1], &b = t_[p << 1 | 1];
56  u.sum = a.sum + b.sum;
57  if (a.mx > b.mx) {
58  u.mx = a.mx; u.cmx = a.cmx;
59  u.mx2 = std::max(a.mx2, b.mx);
60  } else if (a.mx < b.mx) {
61  u.mx = b.mx; u.cmx = b.cmx;
62  u.mx2 = std::max(a.mx, b.mx2);
63  } else {
64  u.mx = a.mx; u.cmx = a.cmx + b.cmx;
65  u.mx2 = std::max(a.mx2, b.mx2);
66  }
67  }

86  void push(int p, int len) {
87  if (len <= 1) return;
88  if (t_[p].lz != T()) {
89  apply_add(p << 1, len - (len >> 1), t_[p].lz);
90  apply_add(p << 1 | 1, len >> 1, t_[p].lz);
91  t_[p].lz = T();
92  }
93  apply_min(p << 1, len - (len >> 1), t_[p].mx);
94  apply_min(p << 1 | 1, len >> 1, t_[p].mx);
95  }

98  void build(int p, int l, int r, const std::vector<T>& a) {
99  t_[p] = Node();
100  if (l == r) {
101  t_[p].mx = t_[p].sum = a[l];
102  t_[p].mx2 = neg_inf_;
103  t_[p].cmx = 1;
104  return;
105  }
106  int mid = (l + r) >> 1;
107  build(p << 1, l, mid, a);
108  build(p << 1 | 1, mid + 1, r, a);
109  pull(p);

113  void add(int p, int l, int r, int ql, int qr, const T& v) {
114  if (ql <= l && r <= qr) { apply_add(p, r - l + 1, v); return; }
115  push(p, r - l + 1);

117  if (ql <= mid) add(p << 1, l, mid, ql, qr, v);
118  if (qr > mid) add(p << 1 | 1, mid + 1, r, ql, qr, v);
119  pull(p);
120  }

123  void chmin(int p, int l, int r, int ql, int qr, const T& x) {
124  if (x >= t_[p].mx) return; // 剪枝：全部无需改
125  if (ql <= l && r <= qr && x > t_[p].mx2) { // 整体取 min 生效
126  apply_min(p, r - l + 1, x);

137  T qsum(int p, int l, int r, int ql, int qr) {
138  if (ql <= l && r <= qr) return t_[p].sum;
139  push(p, r - l + 1);
140  int mid = (l + r) >> 1;
141  T rs = T();
142  if (ql <= mid) rs += qsum(p << 1, l, mid, ql, qr);
143  if (qr > mid) rs += qsum(p << 1 | 1, mid + 1, r, ql, qr);

148  T qmax(int p, int l, int r, int ql, int qr) {
149  if (ql <= l && r <= qr) return t_[p].mx;
150  push(p, r - l + 1);
151  int mid = (l + r) >> 1;
152  T m = neg_inf_;
153  if (ql <= mid) m = std::max(m, qmax(p << 1, l, mid, ql, qr));
154  if (qr > mid) m = std::max(m, qmax(p << 1 | 1, mid + 1, r, ql, qr));
155  return m;
