39 inline bool euler_undirected(const Adj& g, std::vector<int>& path) {
40  int n = (int)g.size() - 1;
41  std::vector<int> deg(n + 1, 0);
42  for (int u = 1; u <= n; ++u) deg[u] = (int)g[u].size();
43  int odd = 0, start = 1;
44  for (int u = 1; u <= n; ++u) {
45  if (deg[u] == 0) continue;
46  start = u;
47  if (deg[u] & 1) { ++odd; start = u; }
48  }
49  if (odd != 0 && odd != 2) return false;
50  // 边多重集（模拟删除；重边用 pair(u,v) 可删一份）
51  int E = 0;
52  std::vector<std::multiset<int>> ms(n + 1);
53  for (int u = 1; u <= n; ++u)
54  for (int v : g[u])
55  if (u < v) { ms[u].insert(v); ms[v].insert(u); ++E; }
56  path.clear();
57  // Hierholzer 递归
58  std::function<void(int)> dfs = [&](int u) {
59  while (!ms[u].empty()) {
60  auto it = ms[u].begin();
61  int v = *it;
62  ms[u].erase(it);
63  auto it2 = ms[v].find(u);
64  ms[v].erase(it2);
65  dfs(v);
66  }
67  path.push_back(u);
68  };
69  dfs(start);
70  if ((int)path.size() != E + 1) return false;
71  std::reverse(path.begin(), path.end());
72  return true;
73 }

81 inline bool euler_directed(const Adj& g, std::vector<int>& path) {
82  int n = (int)g.size() - 1;
83  std::vector<int> indeg(n + 1, 0), outdeg(n + 1, 0);
84  for (int u = 1; u <= n; ++u) {
85  outdeg[u] = (int)g[u].size();
86  for (int v : g[u]) ++indeg[v];
87  }
88  int start = -1, end = -1;
89  for (int u = 1; u <= n; ++u) {
90  int d = outdeg[u] - indeg[u];
91  if (d == 1) { if (start != -1) return false; start = u; }
92  else if (d == -1) { if (end != -1) return false; end = u; }
93  else if (d != 0) return false;
94  }
95  // 选起点
96  std::vector<int> tmp;
97  for (int u = 1; u <= n; ++u) if (outdeg[u] > 0 || indeg[u] > 0) tmp.push_back(u);
98  if (start == -1) start = tmp.empty() ? 1 : tmp[0];
99  // 删边式 DFS：复制出边
100  std::vector<std::multiset<int>> ms(n + 1);
101  for (int u = 1; u <= n; ++u)
102  for (int v : g[u]) ms[u].insert(v);
103  path.clear();
104  std::function<void(int)> dfs = [&](int u) {
105  while (!ms[u].empty()) {
106  auto it = ms[u].begin();
107  int v = *it;
108  ms[u].erase(it);
109  dfs(v);
110  }
111  path.push_back(u);
112  };
113  dfs(start);
114  int total = 0;
115  for (int u = 1; u <= n; ++u) total += outdeg[u];
116  if ((int)path.size() != total + 1) return false;
117  std::reverse(path.begin(), path.end());
118  return true;
119 }
