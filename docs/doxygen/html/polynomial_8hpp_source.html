140 inline std::vector<M> lnf(const std::vector<M>& f, int need) {
141  WBWLIB_ASSERT(!f.empty() && f[0] == M(1));
142  std::vector<M> res(need, M(0));
143  std::vector<M> df = deriv(f); // f'
144  if (df.empty()) return res; // f 为常数 → ln f = 0
145  std::vector<M> fi = invf(f, need); // 1/f
146  std::vector<M> t = mul(df, fi, need - 1); // f'/f，长度 need-1
147  auto iv = inverses(need);
148  for (int i = 1; i < need; ++i) res[i] = t[i - 1] * iv[i]; // 积分
149  return res;
150 }

187 inline std::vector<M> sqrtf(const std::vector<M>& f, int need) {
188  WBWLIB_ASSERT(!f.empty() && f[0] == M(1));
189  static const M inv2 = M(2).inv();
190  std::vector<M> g(1, M(1));
191  int m = 1;
192  while (m < need) {
193  m <<= 1;
194  std::vector<M> gi = invf(g, m); // 1/g
195  std::vector<M> fm = trim(f, m);
196  std::vector<M> t = mul(fm, gi, m); // f/g
197  for (int i = 0; i < m; ++i) {
198  M gv = (i < (int)g.size()) ? g[i] : M(0);
199  t[i] = (t[i] + gv) * inv2; // (f/g + g)/2
200  }
201  g.swap(t);
202  g.resize(m);
203  }
204  g.resize(need);

234 inline void divmod(const std::vector<M>& f, const std::vector<M>& g,
235  std::vector<M>& q, std::vector<M>& r) {
236  int n = (int)f.size(), m = (int)g.size();
237  if (n < m) { q.clear(); r = f; return; }
238  std::vector<M> rf(f.rbegin(), f.rend()); // 系数反转
239  std::vector<M> rgv(g.rbegin(), g.rend());
240  int need = n - m + 1;
241  std::vector<M> ig = invf(trim(rgv, need), need); // 1/reverse(g)
242  std::vector<M> qrev = mul(rf, ig, need);
243  q.assign(qrev.rbegin(), qrev.rend()); // 反转为 q
244  std::vector<M> qg = mul(q, g, n);
245  int rlen = (std::max)(m - 1, 0);
246  r.assign(rlen, M(0));
247  for (int i = 0; i < rlen; ++i) {
248  M a = (i < n) ? f[i] : M(0);
249  M b = (i < (int)qg.size()) ? qg[i] : M(0);
250  r[i] = a - b;
251  }
