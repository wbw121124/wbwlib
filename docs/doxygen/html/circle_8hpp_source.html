58 inline Circle circle_from_three_points(const PointD& a, const PointD& b, const PointD& c) {
59  long double d = (a - c).cross(b - c);
60  long double ax = (a.x - c.x), ay = (a.y - c.y);
61  long double bx = (b.x - c.x), by = (b.y - c.y);
62  long double A = ax * ax + ay * ay;
63  long double B = bx * bx + by * by;
64  long double u = (by * A - ay * B) / (2 * d);
65  long double v = (ax * B - bx * A) / (2 * d);
66  PointD center(u + c.x, v + c.y);
67  return Circle(center, std::sqrt((center - a).norm2()));
68 }

117 inline int circle_relation(const Circle& A, const Circle& B) {
118  long double d = std::sqrt((A.c - B.c).norm2());
119  long double sum = A.r + B.r, dif = std::fabs(A.r - B.r);
120  if (std::fabs(d) < 1e-12L && std::fabs(A.r - B.r) < 1e-12L) return 5;
121  if (d > sum + 1e-12L) return 0;
122  if (std::fabs(d - sum) <= 1e-12L) return 1;
123  if (d > dif + 1e-12L) return 2;
124  if (std::fabs(d - dif) <= 1e-12L) return 3;
125  return 4;
