50  void build(const Adj& g, int root = 1) {
51  int n = (int)g.size() - 1;
52  LOG = 1;
53  while ((1 << LOG) <= n) ++LOG;
54  up_.assign(n + 1, std::vector<int>(LOG, 0));
55  fa.assign(n + 1, 0);
56  dep.assign(n + 1, 0);
57  sz.assign(n + 1, 0);
58  tin.assign(n + 1, 0);
59  tout.assign(n + 1, 0);
60  int timer = 0;
61  std::function<void(int, int)> dfs = [&](int u, int p) {
62  fa[u] = p;
63  dep[u] = dep[p] + 1;
64  sz[u] = 1;
65  tin[u] = ++timer;
66  for (int v : g[u]) {
67  if (v == p) continue;
68  dfs(v, u);
69  sz[u] += sz[v];
70  }
71  tout[u] = ++timer;
72  };
73  dfs(root, 0);
74  for (int u = 1; u <= n; ++u) up_[u][0] = fa[u];
75  for (int j = 1; j < LOG; ++j)
76  for (int u = 1; u <= n; ++u)
77  up_[u][j] = up_[up_[u][j - 1]][j - 1];
78  }

86  bool is_ancestor(int u, int v) const {
87  return tin[u] <= tin[v] && tout[v] <= tout[u];
88  }

96  int query(int u, int v) const {
97  if (is_ancestor(u, v)) return u;
98  if (is_ancestor(v, u)) return v;
99  for (int j = LOG - 1; j >= 0; --j)
100  if (up_[u][j] && !is_ancestor(up_[u][j], v)) u = up_[u][j];
101  return up_[u][0];
102  }

142  std::function<void(int, int, i64)> dfs = [&](int u, int p, i64 d) {
143  dist_to_root[u] = d;
144  for (auto& e : g.g[u])

146  };
147  dfs(root, 0, 0);

156  i64 dist(int u, int v) const {
157  int w = lc.query(u, v);
158  return dist_to_root[u] + dist_to_root[v] - 2 * dist_to_root[w];
