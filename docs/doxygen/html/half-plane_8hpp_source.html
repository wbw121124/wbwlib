62 inline PointD hp_intersect(const HalfPlane& x, const HalfPlane& y) {
63  PointD u = x.b - x.a, v = y.b - y.a;
64  long double den = u.cross(v);
65  // den 不为 0（平行边已剔除）
66  long double t = (y.a - x.a).cross(v) / den;
67  return x.a + u * t;
68 }

75 inline std::vector<PointD> half_plane_intersection(std::vector<HalfPlane> hps) {
76  // 极角增序；同角保留最左（最紧）：更严格的先排，去重时保留第一个
77  std::sort(hps.begin(), hps.end(), [](const HalfPlane& A, const HalfPlane& B) {
78  if (std::fabs(A.ang - B.ang) > 1e-12L) return A.ang < B.ang;
79  return (A.b - A.a).cross(B.a - A.a) < 0; // A 比 B 更靠左（更严格）
80  });
81  std::vector<HalfPlane> uniq;
82  for (auto& hp : hps) {
83  if (!uniq.empty() && std::fabs(hp.ang - uniq.back().ang) <= 1e-12L)
84  continue; // 同极角时 uniq 已保留更严格的
85  uniq.push_back(hp);
86  }
87  std::deque<HalfPlane> q;
88  std::deque<PointD> pts; // q[i] 与 q[i+1] 的交点
89  for (auto& hp : uniq) {
90  while (q.size() >= 2 && !hp.contains(pts.back())) { q.pop_back(); pts.pop_back(); }
91  while (q.size() >= 2 && !hp.contains(pts.front())) { q.pop_front(); pts.pop_front(); }
92  if (!q.empty()) {
93  long double den = (hp.b - hp.a).cross(q.back().b - q.back().a);
94  if (std::fabs(den) <= 1e-12L) {
95  // 平行：若反方向，无解
96  if ((hp.b - hp.a).dot(q.back().b - q.back().a) < 0) return {};
97  continue;
98  }
99  pts.push_back(hp_intersect(q.back(), hp));
100  }
101  q.push_back(hp);
102  }
103  while (q.size() >= 2 && !q.front().contains(pts.back())) { q.pop_back(); pts.pop_back(); }
104  while (q.size() >= 2 && !q.back().contains(pts.front())) { q.pop_front(); pts.pop_front(); }
105  if (q.size() < 3) return {};
106  // 首尾交点
107  std::vector<PointD> res;
108  for (size_t i = 0; i + 1 < q.size(); ++i) res.push_back(pts[i]);
109  res.push_back(hp_intersect(q.back(), q.front()));
110  return res;
111 }
