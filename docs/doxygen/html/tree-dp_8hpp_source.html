38 inline std::tuple<int, int, int> tree_diameter(const Adj& g) {
39  int n = (int)g.size() - 1;
40  std::function<std::pair<int, int>(int)> far = [&](int s) {
41  std::vector<int> dep(n + 1, -1);
42  std::queue<int> q;
43  dep[s] = 0;
44  q.push(s);
45  int best = s;
46  while (!q.empty()) {
47  int u = q.front();
48  q.pop();
49  for (int v : g[u])
50  if (dep[v] < 0) {
51  dep[v] = dep[u] + 1;
52  q.push(v);
53  if (dep[v] > dep[best]) best = v;
54  }
55  }
56  return std::make_pair(best, dep[best]);
57  };
58  int a = far(1).first;
59  auto r = far(a);
60  return std::make_tuple(r.second, a, r.first);
61 }

68 inline std::vector<int> tree_centers(const Adj& g) {
69  int n = (int)g.size() - 1;
70  std::vector<int> sz(n + 1, 0), fa(n + 1, 0);
71  std::vector<int> order;
72  order.reserve(n);
73  std::function<void(int)> dfs = [&](int u) {
74  order.push_back(u);
75  for (int v : g[u])
76  if (v != fa[u]) {
77  fa[v] = u;
78  dfs(v);
79  }
80  };
81  dfs(1);
82  std::vector<int> mx(n + 1, 0);
83  for (int i = n - 1; i >= 0; --i) ++sz[order[i]];
84  int best = n + 1;
85  for (int i = 0; i < n; ++i) {
86  int u = order[i];
87  int cur = n - sz[u];
88  for (int v : g[u])
89  if (v != fa[u]) cur = std::max(cur, sz[v]);
90  mx[u] = cur;
91  best = std::min(best, cur);
92  }
93  std::vector<int> res;
94  for (int u = 1; u <= n; ++u)
95  if (mx[u] == best) res.push_back(u);
96  return res;
97 }

116 std::vector<T> reroot_dp(int n, const Adj& g, int root,
117  T identity, Merge merge, Push push) {
118  std::vector<int> fa(n + 1, 0);
119  std::vector<std::vector<int>> ch(n + 1);
120  std::vector<int> order;
121  order.reserve(n);
122  std::function<void(int)> hang = [&](int u) {
123  order.push_back(u);
124  for (int v : g[u])
125  if (v != fa[u]) {
126  fa[v] = u;
127  ch[u].push_back(v);
128  hang(v);
129  }
130  };
131  hang(root);
132 
133  std::vector<T> dp(n + 1);
134  for (int i = n - 1; i >= 0; --i) {
135  int u = order[i];
136  T acc = identity;
137  for (int v : ch[u]) acc = merge(acc, dp[v]);
138  dp[u] = acc;
139  }
140 
141  std::vector<T> f(n + 1);
142  std::function<void(int, T)> down = [&](int u, T up) {
143  f[u] = merge(up, dp[u]);
144  int k = (int)ch[u].size();
145  // 前缀/后缀
146  std::vector<T> pre(k), suf(k);
147  for (int i = 0; i < k; ++i)
148  pre[i] = merge(i ? pre[i - 1] : identity, dp[ch[u][i]]);
149  for (int i = k - 1; i >= 0; --i)
150  suf[i] = merge(i + 1 < k ? suf[i + 1] : identity, dp[ch[u][i]]);
151  for (int i = 0; i < k; ++i) {
152  T rest = merge(up, merge(i ? pre[i - 1] : identity,
153  i + 1 < k ? suf[i + 1] : identity));
154  down(ch[u][i], push(rest, u, ch[u][i]));
155  }
156  };
157  down(root, identity);
158  return f;
159 }
