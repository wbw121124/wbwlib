63  F dfs(int u, int t, F f) {
64  if (u == t) return f;
65  for (int& i = iter_[u]; i < (int)g_[u].size(); ++i) {
66  Edge& e = g_[u][i];
67  if (e.cap > 0 && level_[e.to] == level_[u] + 1) {
68  F d = dfs(e.to, t, std::min(f, e.cap));
69  if (d > 0) {

73  }
74  }
