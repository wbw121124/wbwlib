49  int newnode(const T& v) {
50  p_.push_back({ {0, 0}, 0, 1, v, v, T(), false });
51  return (int)p_.size() - 1;
52  }

54  // ---------- 懒标记 ----------
56  void push(int x) {
57  if (!x) return;
58  if (p_[x].rev) {
59  std::swap(p_[x].ch[0], p_[x].ch[1]);
60  if (p_[x].ch[0]) p_[p_[x].ch[0]].rev ^= 1;
61  if (p_[x].ch[1]) p_[p_[x].ch[1]].rev ^= 1;
62  p_[x].rev = false;
63  }
64  if (p_[x].add != T()) {
65  T a = p_[x].add;
66  for (int c = 0; c < 2; ++c) if (p_[x].ch[c]) {
67  int u = p_[x].ch[c];
68  p_[u].val += a;
69  p_[u].sum += a * T(p_[u].sz);
70  p_[u].add += a;
71  }
72  p_[x].add = T();
73  }
74  }
75 
77  void pull(int x) {
78  p_[x].sz = 1 + p_[p_[x].ch[0]].sz + p_[p_[x].ch[1]].sz;
79  p_[x].sum = p_[x].val + p_[p_[x].ch[0]].sum + p_[p_[x].ch[1]].sum;
80  }
81 
82  // ---------- 旋转 / 伸展 ----------
84  void rotate(int x) {
85  int y = p_[x].fa, z = p_[y].fa;
86  int k = (p_[y].ch[1] == x);
87  int b = p_[x].ch[k ^ 1];
88  if (z) p_[z].ch[p_[z].ch[1] == y] = x;
89  p_[x].fa = z;
90  p_[x].ch[k ^ 1] = y; p_[y].fa = x;
91  p_[y].ch[k] = b; if (b) p_[b].fa = y;
92  pull(y); pull(x);
93  }
94 
96  void splay(int x) {
97  while (p_[x].fa) {
98  int y = p_[x].fa, z = p_[y].fa;
99  if (z) {
100  if ((p_[y].ch[0] == x) == (p_[z].ch[0] == y)) rotate(y);
101  else rotate(x);
102  }
103  rotate(x);
104  }
105  root_ = x;
106  }
107 
109  int kth(int k) {
110  int x = root_;
111  while (true) {
112  push(x);
113  int lsz = p_[p_[x].ch[0]].sz;
114  if (k == lsz + 1) break;
115  if (k <= lsz) x = p_[x].ch[0];
116  else k -= lsz + 1, x = p_[x].ch[1];
117  }
118  splay(x);
119  return x;
120  }

128  struct Interval {
129  int tree;
130  int parent;
131  int side;
132  };

135  Interval cut(int l, int r) {
136  int n = size();
137  if (l == 1 && r == n) return {root_, 0, -1};
138  Interval iv;
139  if (r < n) {
140  if (l > 1) kth(l - 1);
141  kth(r + 1);
142  iv.parent = root_;
143  iv.side = 0;
144  } else { // r == n
145  kth(l - 1);
146  iv.parent = root_;
147  iv.side = 1;
148  }
149  iv.tree = p_[iv.parent].ch[iv.side];
150  p_[iv.parent].ch[iv.side] = 0;
151  if (iv.tree) p_[iv.tree].fa = 0;
152  pull(iv.parent);
153  return iv;
154  }
155 
157  void restore(const Interval& iv) {
158  if (iv.parent == 0) { root_ = iv.tree; return; }
159  p_[iv.parent].ch[iv.side] = iv.tree;
160  if (iv.tree) p_[iv.tree].fa = iv.parent;
161  pull(iv.parent);
162  root_ = iv.parent;
163  }

181  void build(const std::vector<T>& a) {
182  root_ = 0;
183  if (a.empty()) return;
184  // 平衡建树：取中点递归
185  std::function<int(int, int, int)> rec = [&](int l, int r, int fa) -> int {
186  if (l > r) return 0;
187  int mid = (l + r) >> 1;
188  int x = newnode(a[mid]);
189  p_[x].fa = fa;
190  // 注意：rec 内 newnode 会使 p_ 重分配，必须先拿到返回值再按下标赋值（避免悬空引用）
191  int cl = rec(l, mid - 1, x);
192  int cr = rec(mid + 1, r, x);
193  p_[x].ch[0] = cl;
194  p_[x].ch[1] = cr;
195  pull(x);
196  return x;
197  };
198  root_ = rec(0, (int)a.size() - 1, 0);
199  }

202  int size() const { return p_[root_].sz; }
204  bool empty() const { return root_ == 0; }

211  void insert(int k, const T& v) {
212  int n = size();
213  int x = newnode(v);
214  if (n == 0) { root_ = x; return; }
215  if (k == 1) {
216  p_[x].ch[1] = root_; p_[root_].fa = x;
217  pull(x); root_ = x; return;
218  }
219  if (k == n + 1) {
220  kth(n);
221  p_[root_].ch[1] = x; p_[x].fa = root_;
222  pull(root_); return;
223  }
224  kth(k); // 根为第 k 个；左子树前 k-1 个
225  int L = p_[root_].ch[0];
226  if (L) p_[L].fa = 0;
227  p_[root_].ch[0] = x;
228  p_[x].fa = root_;
229  p_[x].ch[0] = L;
230  if (L) p_[L].fa = x;
231  pull(x); pull(root_);
232  }

244  void erase(int k) {
245  int n = size();
246  if (n == 0) return;
247  if (n == 1) { root_ = 0; return; }
248  kth(k);
249  int L = p_[root_].ch[0], R = p_[root_].ch[1];
250  if (L) p_[L].fa = 0;
251  if (R) p_[R].fa = 0;
252  if (!L) { root_ = R; return; }
253  if (!R) { root_ = L; return; }
254  int m = L;
255  while (p_[m].ch[1]) m = p_[m].ch[1]; // L 中最大
256  while (p_[m].fa) rotate(m); // 旋为 L 子树的根
257  p_[m].ch[1] = R; p_[R].fa = m;
258  pull(m);
259  root_ = m;
260  }
