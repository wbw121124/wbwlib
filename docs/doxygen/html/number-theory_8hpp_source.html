221 inline ExcrtResult excrt(const std::vector<i64>& a, const std::vector<i64>& m) {
222  i64 x = 0, cur = 1; // 已处理方程的通解 x + k*cur（cur 为已处理模数的最小公倍数）
223  for (size_t i = 0; i < a.size(); ++i) {
224  i64 g = gcd(cur, m[i]); // cur 与 m[i] 的公因子
225  i64 q = m[i] / g; // 解 t 的模数
226  i64 diff = ((a[i] - x) % m[i] + m[i]) % m[i];
227  if (diff % g != 0) return {false, 0, 0}; // 无解
228  // 求 t：t ≡ (diff/g) * inv(cur/g, q) (mod q)
229  i64 invc = inv_inline(cur / g, q); // gcd(cur/g, q)=1，逆元必存在
230  i64 t = mul_mod(diff / g, invc, q);
231  t = (t % q + q) % q;
232  x += cur * t;
233  cur *= q; // 更新 lcm
234  x %= cur;
235  }
236  return {true, x, cur};
237 }
