125 std::vector<T> sliding_window(const std::vector<T>& a, int k, Cmp better) {
126  int n = (int)a.size() - 1; // a[0] 占位，数据在 [1..n]
127  std::vector<T> res;
128  res.reserve(n - k + 1);
129  std::deque<int> dq;
130  for (int i = 1; i <= n; ++i) {
131  while (!dq.empty() && better(a[i], a[dq.back()])) dq.pop_back();
132  dq.push_back(i);
133  if (dq.front() <= i - k) dq.pop_front();
134  if (i >= k) res.push_back(a[dq.front()]);
135  }
136  return res;
137 }
