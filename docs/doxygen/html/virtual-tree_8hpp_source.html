50  void build(const std::vector<int>& keys) {
51  const LCA& lc = *lc_;
52  nodes.clear();
53  edges.clear();
54  std::vector<int> ks = keys;
55  std::sort(ks.begin(), ks.end(),
56  [&](int a, int b) { return lc.tin[a] < lc.tin[b]; });
57  ks.erase(std::unique(ks.begin(), ks.end()), ks.end());
58  int k = (int)ks.size();
59  for (int i = k - 1; i >= 1; --i) ks.push_back(lc.query(ks[i - 1], ks[i]));
60  std::sort(ks.begin(), ks.end(),
61  [&](int a, int b) { return lc.tin[a] < lc.tin[b]; });
62  ks.erase(std::unique(ks.begin(), ks.end()), ks.end());
63  nodes = ks;
64  // 栈式建树：弹出时不连边，只在补齐 LCA 层与挂新点时连，最后统一去重
65  std::vector<std::pair<int, int>> es;
66  auto add = [&](int a, int b) {
67  if (a == b) return;
68  es.push_back({a, b});
69  };
70  std::vector<int> st;
71  st.reserve(nodes.size());
72  for (int u : nodes) {
73  if (st.empty()) {
74  st.push_back(u);
75  continue;
76  }
77  int l = lc.query(u, st.back());
78  while ((int)st.size() >= 2 && lc.tin[st[(int)st.size() - 2]] >= lc.tin[l])
79  st.pop_back();
80  if (st.back() != l) {
81  add(l, st.back());
82  st.back() = l;
83  }
84  add(l, u);
85  st.push_back(u);
86  }
87  std::sort(es.begin(), es.end());
88  es.erase(std::unique(es.begin(), es.end()), es.end());
89  edges = es;
90  }
