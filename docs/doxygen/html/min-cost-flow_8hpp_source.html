66  std::pair<i64, i64> min_cost_flow(int s, int t, i64 limit = -1) {
67  const i64 INF = (i64)4e18;
68  i64 flow = 0, cost = 0;
69  while (limit < 0 || flow < limit) {
70  std::vector<i64> dist(n_ + 1, INF);
71  std::vector<char> inq(n_ + 1, 0);
72  std::vector<int> pre(n_ + 1, -1), preE(n_ + 1, -1);
73  std::queue<int> q;
74  dist[s] = 0;
75  inq[s] = 1;
76  q.push(s);
77  while (!q.empty()) {
78  int u = q.front();
79  q.pop();
80  inq[u] = 0;
81  for (int i = 0; i < (int)g_[u].size(); ++i) {
82  const Edge& e = g_[u][i];
83  if (e.cap > 0 && dist[u] + e.cost < dist[e.to]) {
84  dist[e.to] = dist[u] + e.cost;
85  pre[e.to] = u;
86  preE[e.to] = i;
87  if (!inq[e.to]) { inq[e.to] = 1; q.push(e.to); }
88  }
89  }
90  }
91  if (dist[t] >= INF) break;
92  i64 add = limit < 0 ? INF : limit - flow;
93  for (int v = t; v != s; v = pre[v]) add = std::min(add, g_[pre[v]][preE[v]].cap);
94  for (int v = t; v != s; v = pre[v]) {
95  Edge& e = g_[pre[v]][preE[v]];
96  e.cap -= add;
97  g_[v][e.rev].cap += add;
98  }
99  flow += add;
100  cost += add * dist[t];
101  }
102  return {flow, cost};
103  }
104 };
