350 // ---- BigInt 除法：绝对值商二分 ----
351 inline BigInt BigInt::operator/(const BigInt& o) const {
352  if (o.is_zero()) return BigInt(); // 除以 0：返回 0（不抛，OI 场景由用户保证）
353  bool nsign = neg ^ o.neg;
354  const BigInt& a = *this;
355  const BigInt& b = o; // 改成 abs
356  // 构造幅值 a、b
357  BigInt aa = a, bb = b;
358  aa.neg = bb.neg = false;
359  if (aa.abs_cmp(bb) < 0) return BigInt(); // |a| < |b|
360  // 二分答案：商位数为 |a| - |b| + 1
361  int n = (int)aa.d.size() - (int)bb.d.size() + 1;
362  BigInt lo(0), hi(1), mid;
363  hi.d.assign(n + 1, 0);
364  hi.d[n] = 1; // 10^n（BASE^n）
365  // 二分最大 x 使 bb*x <= aa
366  while (lo + (i64)1 < hi) {
367  mid = (lo + hi) / (i64)2;
368  if ((bb * mid).abs_cmp(aa) <= 0) lo = mid;
369  else hi = mid;
370  }
371  BigInt q = lo;
372  q.neg = nsign && !lo.is_zero();
373  q.trim();
374  return q;
375 }
