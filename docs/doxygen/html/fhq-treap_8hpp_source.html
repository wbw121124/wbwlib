69  void split(int x, const T& key, int& l, int& r) {
70  if (!x) { l = r = 0; return; }
71  if (cmp_(pool_[x].val, key)) { // x 的值 < key → 在左侧集合
72  split(pool_[x].r, key, pool_[x].r, r);
73  l = x; pull(l);
74  } else {
75  split(pool_[x].l, key, l, pool_[x].l);
76  r = x; pull(r);
77  }
78  }

81  void split_le(int x, const T& key, int& l, int& r) {
82  if (!x) { l = r = 0; return; }
83  if (cmp_(pool_[x].val, key) || !cmp_(key, pool_[x].val)) { // val <= key
84  split_le(pool_[x].r, key, pool_[x].r, r);
85  l = x; pull(l);
86  } else {
87  split_le(pool_[x].l, key, l, pool_[x].l);
88  r = x; pull(r);
89  }

93  int merge(int l, int r) {
94  if (!l || !r) return l ? l : r;
95  if (pool_[l].pri < pool_[r].pri) {
96  pool_[l].r = merge(pool_[l].r, r);
97  pull(l); return l;
98  } else {
99  pool_[r].l = merge(l, pool_[r].l);
100  pull(r); return r;
101  }

105  int kth(int x, int k) const {
106  while (x) {
107  int lsz = pool_[pool_[x].l].sz;
108  if (k <= lsz) x = pool_[x].l;
109  else if (k == lsz + 1) return x;
110  else { k -= lsz + 1; x = pool_[x].r; }
111  }

219 class ImplicitTreap {
220  struct Node {
221  T val, sum, add;
222  u32 pri;
223  int sz, l, r;
224  bool rev;
225  };
226  std::vector<Node> pool_; // 0 号哑节点，实节点从 1 起
227  int root_ = 0;
228 
230  int newnode(const T& v) {
231  pool_.push_back({v, v, T(), details::randpri(), 1, 0, 0, false});
232  return (int)pool_.size() - 1;
233  }
234 
236  void apply_add(int x, const T& v) {
237  if (!x) return;
238  pool_[x].val += v;
239  pool_[x].sum += v * T(pool_[x].sz);
240  pool_[x].add += v;
241  }
243  void apply_rev(int x) {
244  if (!x) return;
245  pool_[x].rev = !pool_[x].rev;
246  std::swap(pool_[x].l, pool_[x].r);
247  }
249  void push(int x) {
250  if (!x) return;
251  if (pool_[x].add != T()) {
252  apply_add(pool_[x].l, pool_[x].add);
253  apply_add(pool_[x].r, pool_[x].add);
254  pool_[x].add = T();
255  }
256  if (pool_[x].rev) {
257  apply_rev(pool_[x].l);
258  apply_rev(pool_[x].r);
259  pool_[x].rev = false;
260  }
261  }
263  void pull(int x) {
264  pool_[x].sz = 1 + pool_[pool_[x].l].sz + pool_[pool_[x].r].sz;
265  pool_[x].sum = pool_[x].val + pool_[pool_[x].l].sum + pool_[pool_[x].r].sum;
266  }
267 
269  void split(int x, int k, int& l, int& r) {
270  if (!x) { l = r = 0; return; }
271  push(x);
272  if (pool_[pool_[x].l].sz >= k) {
273  split(pool_[x].l, k, l, pool_[x].l);
274  r = x; pull(r);
275  } else {
276  split(pool_[x].r, k - pool_[pool_[x].l].sz - 1, pool_[x].r, r);
277  l = x; pull(l);
278  }
279  }
280 
282  int merge(int l, int r) {
283  if (!l || !r) return l ? l : r;
284  if (pool_[l].pri < pool_[r].pri) {
285  push(l);
286  pool_[l].r = merge(pool_[l].r, r);
287  pull(l); return l;
288  } else {
289  push(r);
290  pool_[r].l = merge(l, pool_[r].l);
291  pull(r); return r;
292  }
293  }

297  if (!x) return;
298  push(x);
299  collect(pool_[x].l, out);
