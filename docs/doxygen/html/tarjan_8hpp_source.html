37 inline int tarjan_scc(const Adj& g, std::vector<int>& scc_id) {
38  int n = (int)g.size() - 1;
39  scc_id.assign(n + 1, 0);
40  std::vector<int> dfn(n + 1, 0), low(n + 1, 0), stk;
41  std::vector<char> in_stk(n + 1, 0);
42  int idx = 0, scc = 0;
43  std::function<void(int)> dfs = [&](int u) {
44  dfn[u] = low[u] = ++idx;
45  stk.push_back(u);
46  in_stk[u] = 1;
47  for (int v : g[u]) {
48  if (!dfn[v]) {
49  dfs(v);
50  low[u] = std::min(low[u], low[v]);
51  } else if (in_stk[v]) {
52  low[u] = std::min(low[u], dfn[v]);
53  }
54  }
55  if (low[u] == dfn[u]) {
56  ++scc;
57  while (true) {
58  int x = stk.back();
59  stk.pop_back();
60  in_stk[x] = 0;
61  scc_id[x] = scc;
62  if (x == u) break;
63  }
64  }
65  };
66  for (int u = 1; u <= n; ++u)
67  if (!dfn[u]) dfs(u);
68  return scc;
69 }

76 inline std::vector<int> tarjan_cut(const Adj& g) {
77  int n = (int)g.size() - 1;
78  std::vector<int> dfn(n + 1, 0), low(n + 1, 0), cut(n + 1, 0);
79  int idx = 0;
80  std::function<void(int, int)> dfs = [&](int u, int fa) {
81  dfn[u] = low[u] = ++idx;
82  int child = 0;
83  for (int v : g[u]) {
84  if (v == fa) continue;
85  if (!dfn[v]) {
86  ++child;
87  dfs(v, u);
88  low[u] = std::min(low[u], low[v]);
89  if (fa != 0 && low[v] >= dfn[u]) cut[u] = 1;
90  } else {
91  low[u] = std::min(low[u], dfn[v]);
92  }
93  }
94  if (fa == 0 && child >= 2) cut[u] = 1;
95  };
96  for (int u = 1; u <= n; ++u)
97  if (!dfn[u]) dfs(u, 0);
98  return cut;
99 }

106 inline void tarjan_bridge(const Adj& g, std::vector<std::pair<int, int>>& bridges) {
107  int n = (int)g.size() - 1;
108  std::vector<int> dfn(n + 1, 0), low(n + 1, 0);
109  int idx = 0;
110  bridges.clear();
111  // 给每条无向边一个 id，建立「点 → (邻点, 边id)」邻接，正确跳过反向重边
112  std::vector<std::vector<std::pair<int, int>>> adj(n + 1);
113  std::vector<int> eu, ev;
114  for (int u = 1; u <= n; ++u)
115  for (int v : g[u]) {
116  if (u >= v) continue;
117  int id = (int)eu.size() + 1;
118  eu.push_back(u);
119  ev.push_back(v);
120  adj[u].push_back({v, id});
121  adj[v].push_back({u, id});
122  }
123  std::function<void(int, int)> dfs = [&](int u, int pe) {
124  dfn[u] = low[u] = ++idx;
125  for (auto& e : adj[u]) {
126  int v = e.first, id = e.second;
127  if (id == pe) continue;
128  if (!dfn[v]) {
129  dfs(v, id);
130  low[u] = std::min(low[u], low[v]);
131  if (low[v] > dfn[u]) bridges.push_back({std::min(u, v), std::max(u, v)});
132  } else {
133  low[u] = std::min(low[u], dfn[v]);
134  }
135  }
136  };
137  for (int u = 1; u <= n; ++u)
138  if (!dfn[u]) dfs(u, 0);
139 }
