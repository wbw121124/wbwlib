53  void build(const Adj& g, int root = 1) {
54  n = (int)g.size() - 1;
55  timer = 0;
56  fa.assign(n + 1, 0);
57  dep.assign(n + 1, 0);
58  sz.assign(n + 1, 0);
59  son.assign(n + 1, 0);
60  top.assign(n + 1, 0);
61  pos.assign(n + 1, 0);
62  order.assign(n + 1, 0);
63  std::vector<int> parent(n + 1, 0);
64  // 第一遍：deep/sz/son
65  std::function<void(int, int)> dfs1 = [&](int u, int p) {
66  fa[u] = p;
67  dep[u] = dep[p] + 1;
68  sz[u] = 1;
69  int best = 0;
70  for (int v : g[u]) {
71  if (v == p) continue;
72  dfs1(v, u);
73  sz[u] += sz[v];
74  if (sz[v] > best) { best = sz[v]; son[u] = v; }
75  }
76  parent[u] = p;
77  };
78  dfs1(root, 0);
79  // 第二遍：top + 重链线性化（重儿子优先），再轻儿子
80  std::function<void(int, int)> dfs2 = [&](int u, int t) {
81  top[u] = t;
82  pos[u] = ++timer;
83  order[timer] = u;
84  if (son[u]) dfs2(son[u], t);
85  for (int v : g[u])
86  if (v != fa[u] && v != son[u]) dfs2(v, v);
87  };
88  dfs2(root, root);
89  }

110  void path(int u, int v, F&& cb) {
111  std::vector<std::pair<int, int>> left, right;
112  while (top[u] != top[v]) {
113  if (dep[top[u]] >= dep[top[v]]) {
114  left.push_back({pos[top[u]], pos[u]});
115  u = fa[top[u]];
116  } else {
117  right.push_back({pos[top[v]], pos[v]});
118  v = fa[top[v]];
119  }
120  }
121  if (dep[u] >= dep[v]) left.push_back({pos[v], pos[u]});
122  else right.push_back({pos[u], pos[v]});
123  // 反转 right，得到 u→v 的正确顺序
124  for (auto& p : left) cb(p.first, p.second);
125  for (int i = (int)right.size() - 1; i >= 0; --i)
126  cb(right[i].first, right[i].second);
127  }

135  int lca(int u, int v) const {
136  while (top[u] != top[v]) {
137  if (dep[top[u]] < dep[top[v]]) std::swap(u, v);
138  u = fa[top[u]];
139  }
140  return dep[u] < dep[v] ? u : v;
141  }
